static void hardResetOLED() {
  pinMode(OLED_RESET, OUTPUT);
  digitalWrite(OLED_RESET, HIGH);
  delay(10);
  digitalWrite(OLED_RESET, LOW);    // Keep low for at least 10ms
  delay(15);
  digitalWrite(OLED_RESET, HIGH);   // Release reset
  delay(50);
}

void I2C_Init(){
  // 1) Turn on I2C peripheral power
  pinMode(PIN_I2C_POWER, OUTPUT);
  digitalWrite(PIN_I2C_POWER, HIGH);
  delay(50);                        // Provide time for power-up reset pin of OLED
  Serial.println(F("I2C power enabled."));
  
  // 2) Start I2C: Use slow speed first to rule out timing issues
  Wire.begin();                     // Use board default SDA/SCL
  Wire.setClock(100000);
  delay(100);                       // Give some time to settle I2C
  Serial.println(F("I2C started at 100kHz."));
}

void OLED_Init(){
  bool ok = display.begin(SSD1306_SWITCHCAPVCC, OLED_ADDR);
  Serial.println(ok ? F("SSD1306 begin (SWITCHCAPVCC) OK")
                    : F("SSD1306 begin (SWITCHCAPVCC) FAIL"));

  if (!ok) {
    hardResetOLED();
    ok = display.begin(SSD1306_EXTERNALVCC, OLED_ADDR);
    Serial.println(ok ? F("SSD1306 begin (EXTERNALVCC) OK")
                      : F("SSD1306 begin (EXTERNALVCC) FAIL"));
  }

  if (ok) {
    display.ssd1306_command(SSD1306_DISPLAYON);
    display.ssd1306_command(SSD1306_SETCONTRAST);
    display.ssd1306_command(0xFF);

    display.clearDisplay();
    display.setTextSize(2);
    display.setTextColor(SSD1306_WHITE);
    display.setCursor(0, 0);
    display.println(F("HELLO"));
    display.drawRect(0, 0, 128, 64, SSD1306_WHITE);
    display.display();
  } else {
    Serial.println(F("OLED init failed."));
  }
}

void ToF_Init(){
  DEV_I2C.begin();
  ToF.begin();
  ToF.VL53L4CX_Off();
  ToF.InitSensor(ToF_ADDR);
  ToF.VL53L4CX_StartMeasurement();
}

void BME_Init(){
  if (!bme.begin(BME_ADDR)) {
    Serial.println("Could not find a valid BME680 sensor, check wiring!");
    while (1);
  }
  // Set up oversampling and filter initialization
  bme.setTemperatureOversampling(BME680_OS_8X);
  bme.setHumidityOversampling(BME680_OS_2X);
  bme.setPressureOversampling(BME680_OS_4X);
  bme.setIIRFilterSize(BME680_FILTER_SIZE_3);
  bme.setGasHeater(320, 150); // 320*C for 150 ms
}

void GPS_Init(){ 
    if (!GPS.begin(GPS_ADDR)) {
      Serial.println("GPS I2C init failed");
    } else {
      // Output only RMC + GGA to save bandwidth; 1Hz update rate
      GPS.sendCommand(PMTK_SET_NMEA_OUTPUT_RMCGGA);
      GPS.sendCommand(PMTK_SET_NMEA_UPDATE_1HZ);
      Serial.println("GPS ready (I2C).");
    }
}

void SD_Init(){
  Serial.print("Initializing SD card...");
  // Note: SD.begin defaults to SPI and configures CS
  if (!SD.begin(SD_CS)) {
    Serial.println("initialization failed!");
    return;
  }
  Serial.println("initialization done.");

  // Open file once and keep it open
  logFile = SD.open("test.txt", FILE_WRITE);
  if (!logFile) {
      Serial.println("Error opening test.txt");
  }
}

// ================= RUNTIME FUNCTIONS (Non-blocking) =================

void ToF_CheckAndRead(){
  VL53L4CX_MultiRangingData_t MultiRangingData;
  VL53L4CX_MultiRangingData_t *pMultiRangingData = &MultiRangingData;
  uint8_t NewDataReady = 0;
  int no_of_object_found = 0, j;
  char report[64];
  int status;

  // Non-blocking check
  status = ToF.VL53L4CX_GetMeasurementDataReady(&NewDataReady);

  if ((!status) && (NewDataReady != 0)) {
    status = ToF.VL53L4CX_GetMultiRangingData(pMultiRangingData);
    no_of_object_found = pMultiRangingData->NumberOfObjectsFound;
    
    snprintf(report, sizeof(report), "ToF: Count=%d, #Objs=%1d ", pMultiRangingData->StreamCount, no_of_object_found);
    SerialPort.print(report);
    
    for (j = 0; j < no_of_object_found; j++) {
      if (j != 0) SerialPort.print(", ");
      SerialPort.print("D=");
      SerialPort.print(pMultiRangingData->RangeData[j].RangeMilliMeter);
      SerialPort.print("mm");
    }
    SerialPort.println("");

    if (status == 0) {
      status = ToF.VL53L4CX_ClearInterruptAndStartMeasurement();
    }
  }
}

void BME_Read(){ 
  if (! bme.performReading()) {
    Serial.println("Failed to perform BME reading :(");
    return;
  }
  Serial.print("Temp: "); Serial.print(bme.temperature); Serial.print(" *C, ");
  Serial.print("Pres: "); Serial.print(bme.pressure / 100.0); Serial.print(" hPa, ");
  Serial.print("Hum: "); Serial.print(bme.humidity); Serial.print(" %, ");
  Serial.print("Gas: "); Serial.print(bme.gas_resistance / 1000.0); Serial.println(" KOhms");
}

void GPS_Service(){
  // This must be called frequently to read characters from I2C buffer
  GPS.read(); 
  
  if (GPS.newNMEAreceived()) {       // Received a complete NMEA sentence
    if (!GPS.parse(GPS.lastNMEA())) return;  // Parse failed, discard
    
    if (GPS.fix) { 
      // Only print/save if fix is valid
      Serial.print("Lat: "); Serial.println(GPS.latitude, 4);
      Serial.print("Lon: "); Serial.println(GPS.longitude, 4);
    }
  }
}

void SD_Service() {
    // 1. Write data to buffer periodically
    if (millis() - lastSdWriteTime > INTERVAL_LOG) {
        lastSdWriteTime = millis();
        if (logFile) {
            // Example: Write timestamp
            logFile.print(millis());
            logFile.println(", testing 1, 2, 3.");
        }
    }

    // 2. Flush data to card physically periodically (prevents blocking every loop)
    if (millis() - lastSdFlushTime > INTERVAL_SD_FLUSH) {
        lastSdFlushTime = millis();
        if (logFile) {
            logFile.flush();
            Serial.println("SD Flushed.");
        }
    }
}

// ================= SETUP & LOOP =================

void setupPeripherals() {
  Serial.println(F("Setup peripherals started."));
  I2C_Init();
  OLED_Init();
  SD_Init();
  GPS_Init();
  BME_Init();
  ToF_Init();
}

void serviceOtherPeripheralsNonBlocking() {
  unsigned long now = millis();

  // 1. High frequency polling (GPS needs this)
  GPS_Service();

  // 2. ToF (Fast polling)
  if (now - lastTofTime > INTERVAL_TOF) {
      lastTofTime = now;
      ToF_CheckAndRead();
  }

  // 3. BME Sensor (Slow polling)
  if (now - lastBmeTime > INTERVAL_BME) {
      lastBmeTime = now;
      BME_Read();
  }

  // 4. Display Update
  if (now - lastDisplayTime > INTERVAL_DISPLAY) {
      lastDisplayTime = now;
      Display_update();
  }

  // 5. SD Card Service
  SD_Service();
}

void setup() {
  Serial.begin(115200);
  delay(100); // Wait for serial
  
  setupPeripherals();

}
