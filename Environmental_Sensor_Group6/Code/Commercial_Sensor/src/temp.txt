// ====== Settings and Main Loop ======
void setup() {
  Serial.begin(115200);


  if (!display.begin(SSD1306_SWITCHCAPVCC, SCREEN_ADDRESS)) {
    Serial.println(F("SSD1306 allocation failed"));
    for (;;)
      delay(10);
  }

  display.display();
  delay(1000);
  display.clearDisplay();

  display.drawPixel(10, 10, SSD1306_WHITE);
  display.display();
  delay(1000);

  testdrawline();
  testdrawrect();
  testfillrect();
  testdrawcircle();
  testfillcircle();
  testdrawroundrect();
  testfillroundrect();
  testdrawtriangle();
  testfilltriangle();
  testdrawchar();
  testdrawstyles();
  testscrolltext();
  testdrawbitmap();

  display.invertDisplay(true);  delay(500);
  display.invertDisplay(false); delay(500);

  testanimate(logo_bmp, LOGO_WIDTH, LOGO_HEIGHT);
}

void testdrawline() {
  int16_t i;
  display.clearDisplay();

  for (i = 0; i < display.width(); i += 4) {
    display.drawLine(0, 0, i, display.height() - 1, SSD1306_WHITE);
    display.display();
    delay(1);
  }
  for (i = 0; i < display.height(); i += 4) {
    display.drawLine(0, 0, display.width() - 1, i, SSD1306_WHITE);
    display.display();
    delay(1);
  }
  delay(250);

  display.clearDisplay();
  for (i = 0; i < display.width(); i += 4) {
    display.drawLine(0, display.height() - 1, i, 0, SSD1306_WHITE);
    display.display();
    delay(1);
  }
  for (i = display.height() - 1; i >= 0; i -= 4) {
    display.drawLine(0, display.height() - 1, display.width() - 1, i, SSD1306_WHITE);
    display.display();
    delay(1);
  }
  delay(250);

  display.clearDisplay();
  for (i = display.width() - 1; i >= 0; i -= 4) {
    display.drawLine(display.width() - 1, display.height() - 1, i, 0, SSD1306_WHITE);
    display.display();
    delay(1);
  }
  for (i = display.height() - 1; i >= 0; i -= 4) {
    display.drawLine(display.width() - 1, display.height() - 1, 0, i, SSD1306_WHITE);
    display.display();
    delay(1);
  }
  delay(250);

  display.clearDisplay();
  for (i = 0; i < display.height(); i += 4) {
    display.drawLine(display.width() - 1, 0, 0, i, SSD1306_WHITE);
    display.display();
    delay(1);
  }
  for (i = 0; i < display.width(); i += 4) {
    display.drawLine(display.width() - 1, 0, i, display.height() - 1, SSD1306_WHITE);
    display.display();
    delay(1);
  }
  delay(1000);
}

#define XPOS   0
#define YPOS   1
#define DELTAY 2

void testanimate(const uint8_t *bitmap, uint8_t w, uint8_t h) {
  int8_t f, icons[NUMFLAKES][3];

  for (f = 0; f < NUMFLAKES; f++) {
    icons[f][XPOS]   = random(1 - (int)LOGO_WIDTH, display.width());
    icons[f][YPOS]   = -LOGO_HEIGHT;
    icons[f][DELTAY] = random(1, 6);
    Serial.print(F("x: "));  Serial.print(icons[f][XPOS], DEC);
    Serial.print(F(" y: ")); Serial.print(icons[f][YPOS], DEC);
    Serial.print(F(" dy: "));Serial.println(icons[f][DELTAY], DEC);
  }

  for (;;) {
    display.clearDisplay();
    for (f = 0; f < NUMFLAKES; f++) {
      display.drawBitmap(icons[f][XPOS], icons[f][YPOS], bitmap, w, h, SSD1306_WHITE);
    }
    display.display();
    delay(200);

    for (f = 0; f < NUMFLAKES; f++) {
      icons[f][YPOS] += icons[f][DELTAY];
      if (icons[f][YPOS] >= display.height()) {
        icons[f][XPOS]   = random(1 - (int)LOGO_WIDTH, display.width());
        icons[f][YPOS]   = -LOGO_HEIGHT;
        icons[f][DELTAY] = random(1, 6);
      }
    }
  }
}




#include <Arduino.h>
#include <Wire.h>


#define OLED_ADDR 0x3D

#ifndef PIN_I2C_POWER
#define PIN_I2C_POWER 7 
#endif

void scan_i2c() {
  Serial.println(F("\n[I2C] Scanning..."));
  uint8_t found = 0;

  for (uint8_t addr = 1; addr < 127; addr++) {
    Wire.beginTransmission(addr);
    uint8_t err = Wire.endTransmission();   // 0=ACK
    if (err == 0) {
      Serial.print(F("  - found device at 0x"));
      if (addr < 16) Serial.print("0");
      Serial.print(addr, HEX);
      Serial.println();
      found++;
    }
    delay(2);
  }
  if (!found) Serial.println(F("  (no I2C devices found)"));
}

bool read_max17048_demo() {
  const uint8_t kAddr = 0x36;          // MAX17048/49
  Wire.beginTransmission(kAddr);
  Wire.write(0x02);                     // VCELL register MSB
  if (Wire.endTransmission(false) != 0) return false;
  if (Wire.requestFrom((int)kAddr, 2) != 2) return false;
  uint16_t raw = (Wire.read() << 8) | Wire.read();
  // MAX17048: 78.125 uV/LSB，Calculate the voltage (approximate value).
  float voltage = (raw >> 4) * 0.00125f; // 1.25 mV/LSB
  Serial.print(F("[MAX17048] VCELL ≈ "));
  Serial.print(voltage, 3);
  Serial.println(F(" V"));
  return true;
}

void setup() {
  Serial.begin(115200);
  while (!Serial) { delay(10); }

  // 1) Turn on STEMMA QT I2C power supply
  pinMode(PIN_I2C_POWER, OUTPUT);
  digitalWrite(PIN_I2C_POWER, HIGH);
  delay(10);


  Serial.println(F("\n===== I2C bring-up ====="));
  Serial.print(F("SDA pin = ")); Serial.println(SDA);
  Serial.print(F("SCL pin = ")); Serial.println(SCL);

  // 3) scanning
  scan_i2c();


  read_max17048_demo();


  Serial.print(F("Expect OLED at 0x"));
  Serial.println(OLED_ADDR, HEX);
  Serial.println(F("If not found: check 3V3/GND/SDA/SCL & QT power pin."));
}

void loop() {
  static uint32_t t0 = 0;
  if (millis() - t0 > 5000) {
    t0 = millis();
    scan_i2c();
  }
}
